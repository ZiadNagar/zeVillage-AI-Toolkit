# JSXGraph 3D — Deep Reference

JSXGraph 3D is fully built-in since v1.4. All 3D elements live inside a `View3D` element
which handles projection (oblique, orthographic, or perspective).

---

## Setup

```js
const board = JXG.JSXGraph.initBoard('jxgbox', {
  boundingbox: [-6, 6, 6, -6],
  keepaspectratio: true,
  showNavigation: false  // 3D has its own navigation
});

const view = board.create('view3d',
  [
    [-4, -3],              // [x2D, y2D] lower-left corner of 3D frame in board units
    [8, 8],                // [width, height] of 3D frame in board units
    [[-5,5], [-5,5], [-5,5]]  // [[xRange], [yRange], [zRange]] 3D bounding box
  ],
  {
    projection: 'central',  // 'central' (perspective) | 'parallel' (orthographic)
    xPlaneRear:  { fillColor: '#dde', fillOpacity: 0.3, visible: true },
    yPlaneRear:  { fillColor: '#ded', fillOpacity: 0.3, visible: true },
    zPlaneRear:  { fillColor: '#edd', fillOpacity: 0.3, visible: true },
    xPlaneFront: { visible: false },
    yPlaneFront: { visible: false },
    zPlaneFront: { visible: false },
  }
);
```

### Camera / Azimuth / Elevation
```js
view.D3.az_slide;   // azimuth slider (auto-created if showAzimuthBar: true)
view.D3.el_slide;   // elevation slider

// Set camera angles programmatically
view.setView(az, el, r); // az=azimuth°, el=elevation°, r=zoom
```

## 3D Elements (call via `view.create(...)`)

### Axes3D
```js
view.create('axes3d', [], {
  xAxis: { strokeColor: 'red',   name: 'x', withLabel: true },
  yAxis: { strokeColor: 'green', name: 'y', withLabel: true },
  zAxis: { strokeColor: 'blue',  name: 'z', withLabel: true }
});
```

### Point3D
```js
const p = view.create('point3d', [1, 2, 3], { name: 'P', size: 4 });
// Dynamic:
view.create('point3d', [() => t, () => Math.sin(t), 0]);
// Methods: p.X(), p.Y(), p.Z()
```

### Line3D
```js
view.create('line3d', [p1, p2], { strokeColor: '#333' });
view.create('line3d', [[0,0,0], [1,1,1]], { strokeWidth: 2 });
```

### Curve3D
```js
// Parametric space curve
view.create('curve3d',
  [t => Math.cos(t), t => Math.sin(t), t => t * 0.3, 0, 6 * Math.PI],
  { strokeColor: '#c00', strokeWidth: 2, numberPoints: 200 }
);
```

### Functiongraph3D — z = f(x,y)
```js
view.create('functiongraph3d',
  [(x, y) => Math.sin(Math.sqrt(x*x + y*y)) / Math.sqrt(x*x + y*y + 0.01)],
  {
    xRange: [-5, 5],
    yRange: [-5, 5],
    stepsX: 50,
    stepsY: 50,
    strokeWidth: 0.5,
    fillColor: '#4a9',
    fillOpacity: 0.8
  }
);
```

### ParametricSurface3D — [x(u,v), y(u,v), z(u,v)]
```js
// Torus
view.create('parametricsurface3d',
  [
    (u, v) => (3 + Math.cos(v)) * Math.cos(u),
    (u, v) => (3 + Math.cos(v)) * Math.sin(u),
    (u, v) => Math.sin(v)
  ],
  {
    uRange: [0, 2 * Math.PI],
    vRange: [0, 2 * Math.PI],
    stepsU: 48,
    stepsV: 24,
    fillColor: '#c66',
    fillOpacity: 0.7,
    strokeWidth: 0.3
  }
);
```

### Sphere3D
```js
view.create('sphere3d', [centerPoint3D, radius], { fillColor: '#88f', fillOpacity: 0.4 });
```

### Plane3D
```js
// Plane through 3 points
view.create('plane3d', [p1, p2, p3], { fillColor: '#aaf', fillOpacity: 0.3 });

// Plane by point + two direction vectors
view.create('plane3d',
  [[0,0,0], [1,0,0], [0,1,0]],
  { xRange: [-3,3], yRange: [-3,3] }
);
```

### Polygon3D
```js
view.create('polygon3d', [p1, p2, p3, p4], { fillColor: '#8a8', fillOpacity: 0.5 });
```

### Circle3D
```js
// Circle in 3D: center, normal direction, radius
view.create('circle3d',
  [centerPt3D, [0, 0, 1], 2],  // center, normalVector, radius
  { strokeColor: '#c00', strokeWidth: 2 }
);
```

### Text3D
```js
view.create('text3d', [x, y, z, 'Label'], { fontSize: 14, color: '#333' });
```

### Vectorfield3D
```js
view.create('vectorfield3d',
  [(x,y,z) => [-y, x, 0.5]],  // returns [vx, vy, vz]
  { scale: 0.3, gridX: 5, gridY: 5, gridZ: 3 }
);
```

### Mesh3D
```js
view.create('mesh3d',
  [(u,v) => u, (u,v) => v, (u,v) => 0, -3, 3, -3, 3],
  { strokeColor: '#aaa', strokeWidth: 0.5 }
);
```

## Interactivity with 3D

```js
// Draggable 3D point (constrained to XY plane by default)
const drag = view.create('point3d', [1, 1, 0], {
  fixed: false,  // allow dragging in 2D projection
  name: 'Drag me'
});

// Glider on 3D curve
const g3 = view.create('glider3d', [0, curve3d]);

// Animation
let angle = 0;
setInterval(() => {
  angle += 0.02;
  drag.setPosition(JXG.COORDS_BY_USER, [Math.cos(angle), Math.sin(angle), 0]);
  board.update();
}, 30);
```

## Complete Example — Interactive 3D Sine Surface

```js
const board = JXG.JSXGraph.initBoard('jxgbox', {
  boundingbox: [-5, 5, 5, -5],
  keepaspectratio: true
});

const view = board.create('view3d',
  [[-4,-3], [8,8], [[-4,4],[-4,4],[-2,2]]],
  { projection: 'central' }
);

view.create('axes3d', [], {
  xAxis: { strokeColor: 'red' },
  yAxis: { strokeColor: 'green' },
  zAxis: { strokeColor: 'blue' }
});

const aSlider = board.create('slider',
  [[-4,-4.5],[4,-4.5],[0.1, 1, 3]],
  { name: 'a', label: { fontSize: 14 } }
);

view.create('functiongraph3d',
  [(x,y) => aSlider.Value() * Math.sin(Math.sqrt(x*x + y*y))],
  { xRange: [-4,4], yRange: [-4,4], stepsX: 40, stepsY: 40,
    fillColor: '#4466cc', fillOpacity: 0.7 }
);
```
